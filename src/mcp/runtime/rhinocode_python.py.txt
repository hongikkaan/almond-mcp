def _diagnostic_text(text, maximum, tail=False):
    def size(value):
        return len(json.dumps(value, ensure_ascii=True).encode("ascii"))
    if size(text) <= maximum:
        return text
    marker = "... [truncated] ..."
    def clipped(count):
        if tail:
            return marker + (text[-count:] if count else "")
        return text[:count] + marker
    low, high = 0, len(text)
    while low < high:
        middle = (low + high + 1) // 2
        if size(clipped(middle)) <= maximum:
            low = middle
        else:
            high = middle - 1
    return clipped(low)


def _error_text(error, limit):
    try:
        try:
            text_type = unicode
        except NameError:
            text_type = str
        return _diagnostic_text(text_type(error)[:limit], limit)
    except BaseException:
        return _diagnostic_text("Exception text could not be converted (" +
                                type(error).__name__[:128] + ").", limit)


def _error_trace():
    try:
        return _diagnostic_text(traceback.format_exc()[-8000:], 8000, tail=True)
    except BaseException:
        return "Exception traceback could not be formatted."


def _atomic_json(path, value, maximum):
    temporary = path + ".new"
    length = 0
    digest = hashlib.sha256()
    created = False
    try:
        descriptor = os.open(temporary, os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o600)
        created = True
        with os.fdopen(descriptor, "wb") as stream:
            encoder = json.JSONEncoder(ensure_ascii=True, allow_nan=False, separators=(",", ":"))
            for piece in encoder.iterencode(value):
                data = piece.encode("ascii")
                length += len(data)
                if length > maximum:
                    raise ValueError("JSON exceeds the retained result limit.")
                stream.write(data)
                digest.update(data)
            stream.flush()
        # Request files are owned and published once. rename works
        # on Python 2 and never replaces a receipt from another execution.
        os.rename(temporary, path)
    finally:
        if created and os.path.exists(temporary):
            os.remove(temporary)
    return length, digest.hexdigest()


def _publish_completion(directory, final):
    path = os.path.join(directory, "complete.json")
    try:
        _atomic_json(path, final, 65536)
    except ValueError:
        # A valid nested UTF-8 receipt can grow when re-encoded as ASCII JSON.
        # Keep identity, execution/mutation state, Undo and result proofs intact.
        final["diagnostics_truncated"] = True
        for field in ("error", "result_error"):
            if field not in final:
                continue
            previous = final[field]
            final[field] = dict(message="Diagnostic details omitted to fit the completion receipt.")
            if isinstance(previous, dict):
                for key in ("stage", "type"):
                    if key in previous:
                        final[field][key] = _error_text(previous[key], 128)
        if final["cleanup_errors"]:
            final["cleanup_errors"] = ["Cleanup diagnostics omitted to fit the completion receipt."]
        final.pop("child_undo", None)
        _atomic_json(path, final, 65536)


class _Log(object):
    encoding = "utf-8"
    errors = "replace"

    def __init__(self, path):
        self.stream = io.open(path, "wb")
        self.retained = 0
        self.total = 0

    def write(self, value):
        try:
            text = value if isinstance(value, type(u"")) else value.decode("utf-8", "replace")
        except AttributeError:
            text = str(value)
        data = text.encode("utf-8", "replace")
        available = max(0, 1048576 - self.retained)
        self.stream.write(data[:available])
        self.stream.flush()
        self.retained += min(available, len(data))
        self.total += len(data)
        return len(value)

    def flush(self):
        self.stream.flush()

    def isatty(self):
        return False

    def close(self):
        self.stream.close()
